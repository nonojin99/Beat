currentSource = audioCtx.createBufferSource();
currentSource.buffer = audioBuffer;
currentSource.connect(audioCtx.destination);
currentSource.start(0);
console.log('🔊 Web Audio API로 음원 정상 재생!');
})
.catch(function(err) {
console.error('오디오 디코딩 에러:', err);
});

startTime = performance.now();
isPlaying = true;
requestAnimationFrame(gameLoop);
}

function generateChart(bpm, durationSec) {
var notes = [];
var interval = 60 / bpm;
var currentTime = 2.0;
var totalBeats = Math.floor(durationSec / interval);

for (var i = 0; i < totalBeats; i++) {
var lane = Math.floor(Math.random() * 4);
notes.push({ time: currentTime, lane: lane, hit: false, missed: false });

if (selectedDiff.level > 5 && Math.random() > 0.5) {
var secondLane = (lane + 1 + Math.floor(Math.random() * 3)) % 4;
notes.push({ time: currentTime, lane: secondLane, hit: false, missed: false });
}
currentTime += (Math.random() > 0.2 ? interval : interval * 0.5);
}
return notes;
}

function gameLoop(timestamp) {
if (!isPlaying) return;

var elapsedTime = (timestamp - startTime) / 1000;
var speed = selectedDiff.speed;
var judgmentY = canvas.height - 120;

activeNotes.forEach(function(note) {
if (!note.hit && !note.missed) {
var noteY = judgmentY - (note.time - elapsedTime) * speed;
if (noteY > judgmentY + 80) {
note.missed = true;
handleJudgment('MISS');
}
}
});

document.getElementById('hudScore').innerText = score.toLocaleString();
var acc = totalNotes > 0 ? Math.round((perfectCount / totalNotes) * 100) : 100;
document.getElementById('hudAccuracy').innerText = acc + '%';

var remaining = activeNotes.filter(function(n) { return !n.hit && !n.missed; });
if (remaining.length === 0 && elapsedTime > (activeNotes[activeNotes.length - 1].time + 2)) {
endGame();
return;
}

renderGame(elapsedTime, judgmentY, speed);
requestAnimationFrame(gameLoop);
}

function renderGame(elapsedTime, judgmentY, speed) {
ctx.clearRect(0, 0, canvas.width, canvas.height);

var stageWidth = Math.min(500, canvas.width * 0.4);
var stageLeft = (canvas.width - stageWidth) / 2;
var laneWidth = stageWidth / 4;

ctx.fillStyle = 'rgba(17, 19, 31, 0.85)';
ctx.fillRect(stageLeft, 0, stageWidth, canvas.height);

for (var i = 0; i < 4; i++) {
var x = stageLeft + i * laneWidth;
if (KEY_STATE[i]) {
ctx.fillStyle = 'rgba(0, 242, 254, 0.15)';
ctx.fillRect(x, 0, laneWidth, canvas.height);
}
ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
ctx.strokeRect(x, 0, laneWidth, canvas.height);
}

ctx.strokeStyle = '#00f2fe';
ctx.lineWidth = 4;
ctx.shadowColor = '#00f2fe';
ctx.shadowBlur = 15;
ctx.beginPath();
ctx.moveTo(stageLeft, judgmentY);
ctx.lineTo(stageLeft + stageWidth, judgmentY);
ctx.stroke();
ctx.shadowBlur = 0;

activeNotes.forEach(function(note) {
if (!note.hit && !note.missed) {
var noteY = judgmentY - (note.time - elapsedTime) * speed;
if (noteY > -50 && noteY < canvas.height + 50) {
var nx = stageLeft + note.lane * laneWidth + 6;
var nw = laneWidth - 12;

ctx.fillStyle = LANE_COLORS[note.lane];
ctx.shadowColor = LANE_COLORS[note.lane];
ctx.shadowBlur = 10;

if (typeof ctx.roundRect === 'function') {
ctx.beginPath();
ctx.roundRect(nx, noteY - 10, nw, 20, 6);
ctx.fill();
} else {
ctx.fillRect(nx, noteY - 10, nw, 20);
}
ctx.shadowBlur = 0;
}
}
});

for (var j = 0; j < 4; j++) {
var kx = stageLeft + j * laneWidth;
if (KEY_STATE[j]) {
ctx.fillStyle = LANE_COLORS[j];
ctx.globalAlpha = 0.5;
ctx.fillRect(kx + 2, judgmentY, laneWidth - 4, canvas.height - judgmentY);
ctx.globalAlpha = 1.0;
}
ctx.fillStyle = '#ffffff';
ctx.font = 'bold 22px Arial';
ctx.textAlign = 'center';
ctx.fillText(LANE_NAMES[j], kx + laneWidth / 2, canvas.height - 40);
}

if (judgmentTimer > 0) {
judgmentTimer--;
ctx.font = 'black 36px Arial';
ctx.textAlign = 'center';
ctx.fillStyle = lastJudgment === 'PERFECT' ? '#00f2fe' : (lastJudgment === 'GREAT' ? '#ffbe0b' : '#ff0055');
ctx.fillText(lastJudgment, canvas.width / 2, judgmentY - 80);

if (combo > 1) {
ctx.font = 'bold 24px Arial';
ctx.fillStyle = '#ffffff';
ctx.fillText(combo + ' COMBO', canvas.width / 2, judgmentY - 40);
}
}
}

window.addEventListener('keydown', function(e) {
var idx = LANE_KEYS.indexOf(e.code);
if (idx !== -1 && !KEY_STATE[idx]) {
KEY_STATE[idx] = true;
if (isPlaying) processHit(idx);
}
});

window.addEventListener('keyup', function(e) {
var idx = LANE_KEYS.indexOf(e.code);
if (idx !== -1) KEY_STATE[idx] = false;
});

function processHit(lane) {
var currentTime = (performance.now() - startTime) / 1000;
var target = null;
var minDiff = Infinity;

activeNotes.forEach(function(note) {
if (note.lane === lane && !note.hit && !note.missed) {
var diff = Math.abs(note.time - currentTime);
if (diff < minDiff) {
minDiff = diff;
target = note;
}
}
});

if (target && minDiff <= 0.14) {
target.hit = true;
if (minDiff <= 0.04) handleJudgment('PERFECT');
else if (minDiff <= 0.08) handleJudgment('GREAT');
else handleJudgment('GOOD');
}
}

function handleJudgment(type) {
lastJudgment = type;
judgmentTimer = 30;
totalNotes++;

if (type === 'PERFECT') {
score += 1000 + combo * 10;
combo++;
perfectCount++;
} else if (type === 'GREAT') {
score += 700 + combo * 5;
combo++;
} else if (type === 'GOOD') {
score += 300;
combo++;
} else {
combo = 0;
}
if (combo > maxCombo) maxCombo = combo;
}

function endGame() {
isPlaying = false;
stopAudio();

var track = TRACKS[currentTrackIndex];
document.getElementById('resultSongInfo').innerText = track.title + ' [' + selectedDiff.name + ']';
document.getElementById('resScore').innerText = score.toLocaleString();
document.getElementById('resCombo').innerText = maxCombo;
var acc = totalNotes > 0 ? (perfectCount / totalNotes * 100).toFixed(1) : 100;
document.getElementById('resAccuracy').innerText = acc + '%';

saveAndRenderLeaderboard(track.id, score, maxCombo);
switchScreen('resultScreen');
}

async function saveAndRenderLeaderboard(songId, finalScore, maxCombo) {
var rows = document.getElementById('leaderboardRows');
rows.innerHTML = '<tr><td colspan="4">Loading Leaderboard...</td></tr>';

if (supabaseClient) {
try {
await supabaseClient.from('leaderboard').insert([{
user_name: currentUser,
song_id: songId,
score: finalScore,
max_combo: maxCombo
}]);

var res = await supabaseClient
.from('top10_leaderboard')
.select('*')
.eq('song_id', songId)
.lte('rank', 10);

if (res.data && res.data.length > 0) {
renderLeaderboardTable(res.data);
return;
}
} catch (e) {
console.error('Supabase Sync Error:', e);
}
}

var dummyData = [
{ user_name: 'ProRhythm', score: finalScore + 5200, max_combo: maxCombo + 10 },
{ user_name: currentUser, score: finalScore },
{ user_name: 'BeatMaster', score: Math.max(0, finalScore - 3000), max_combo: Math.max(0, maxCombo - 15) }
 ].sort(function(a, b) { return b.score - a.score; });

renderLeaderboardTable(dummyData);
}

function renderLeaderboardTable(data) {
var rows = document.getElementById('leaderboardRows');
rows.innerHTML = '';

data.forEach(function(item, idx) {
var tr = document.createElement('tr');
if (item.user_name === currentUser) tr.className = 'highlight';

var rankClass = idx === 0 ? 'rank-1' : (idx === 1 ? 'rank-2' : (idx === 2 ? 'rank-3' : ''));

tr.innerHTML =
'<td><span class="rank-badge ' + rankClass + '">' + (idx + 1) + '</span></td>' +
'<td>' + item.user_name + '</td>' +
'<td>' + item.score.toLocaleString() + '</td>' +
'<td>' + item.max_combo + '</td>';
rows.appendChild(tr);
});
}
